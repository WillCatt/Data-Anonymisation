# Dedicated requirements for the local Gradio demo.
#
# The main repo venv uses spaCy 3.7, transformers 4.40+, and recent
# huggingface-hub — all of which conflict with newer Gradio releases.
# This file pins a set of versions known to coexist cleanly so you can
# run the Gradio demo in its own venv without touching the main one.
#
# Bootstrap with `demo/run_gradio.sh`.

# Gradio 4.x — the line our demo/app.py was written against.
# 5.x and 6.x have API-breaking changes (Textbox kwargs, theme location).
gradio>=4.20.0,<5.0.0

# huggingface_hub — Gradio 4.x imports HfFolder which was removed in 0.27.
huggingface-hub>=0.19.0,<0.27.0

# Typer pin — spaCy 3.7 requires typer<0.10. Newer Gradio installs pulled
# in typer 0.25 which breaks `python -m spacy download`. Pin it back.
typer>=0.3.0,<0.10.0

# spaCy for the NER backbone. Pin to the main-venv version so any model
# we've already downloaded works without re-downloading.
spacy==3.7.4

# The sm model as a direct wheel — `python -m spacy download` can 404 when
# spaCy's compatibility-table lookup resolves to an empty version. Pinning the
# wheel makes the bootstrap deterministic (run_gradio.sh no longer needs the
# separate `spacy download` step).
https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.7.1/en_core_web_sm-3.7.1-py3-none-any.whl

# Dataset loading for the mosaic scorer (Phase 3 Pro pipeline).
datasets==2.18.0
pandas>=2.0.0
